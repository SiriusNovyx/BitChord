cmake_minimum_required(VERSION 3.20)
project(bitchord_webview LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    message(FATAL_ERROR "WebView2 host requires MSVC ARM64")
endif()
if(NOT EXISTS "${WEBVIEW2_SDK}/build/native/arm64/WebView2LoaderStatic.lib")
    message(FATAL_ERROR "Set WEBVIEW2_SDK to the extracted Microsoft.Web.WebView2 NuGet package")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(bitchord_webview SHARED webview_jni.cpp)
set_target_properties(bitchord_webview PROPERTIES CXX_STANDARD 17 PREFIX "" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
target_include_directories(bitchord_webview PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/win32" "${WEBVIEW2_SDK}/build/native/include")
target_compile_definitions(bitchord_webview PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(bitchord_webview PRIVATE "${WEBVIEW2_SDK}/build/native/arm64/WebView2LoaderStatic.lib" ole32 user32 shlwapi version)
