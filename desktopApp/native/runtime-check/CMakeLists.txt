cmake_minimum_required(VERSION 3.20)
project(bitchord_runtime_check LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    message(FATAL_ERROR "Runtime compatibility check requires MSVC ARM64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(bitchord_runtime_check SHARED runtime_check.cpp)
set_target_properties(bitchord_runtime_check PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
target_include_directories(bitchord_runtime_check PRIVATE "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/win32")
target_compile_definitions(bitchord_runtime_check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
